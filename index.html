const http = require('http'); const socketIo = require('socket.io'); const multer = require('multer'); const path = require('path'); const fs = require('fs'); const app = express(); const server = http.createServer(app); const io = socketIo(server); // إعداد المجلدات const videosDir = './videos'; if (!fs.existsSync(videosDir)) fs.mkdirSync(videosDir); app.use(express.static('public')); app.use('/videos', express.static(videosDir)); // إعداد رفع الفيديوهات const storage = multer.diskStorage({ destination: videosDir, filename: (req, file, cb) => { cb(null, Date.now() + '-' + file.originalname); } }); const upload = multer({ storage }); // قاعدة بيانات مؤقتة (في الذاكرة) let posts = []; // { id, text, imageUrl, videoUrl, likes, comments, timestamp } let chatMessages = []; // { user, message, timestamp } let users = {}; // socketId -> username // API: رفع فيديو (تيك توك) app.post('/upload-video', upload.single('video'), (req, res) => { const videoUrl = `/videos/${req.file.filename}`; const newPost = { id: Date.now(), text: req.body.text || '', videoUrl: videoUrl, likes: 0, comments: [], timestamp: new Date().toISOString() }; posts.unshift(newPost); io.emit('new-post', newPost); res.json({ success: true, post: newPost }); }); // API: رفع منشور نصي (فيسبوك/تويتر) app.post('/upload-text', express.json(), (req, res) => { const newPost = { id: Date.now(), text: req.body.text, imageUrl: req.body.imageUrl || null, likes: 0, comments: [], timestamp: new Date().toISOString() }; posts.unshift(newPost); io.emit('new-post', newPost); res.json({ success: true, post: newPost }); }); // API: الحصول على جميع المنشورات app.get('/posts', (req, res) => { res.json(posts); }); // API: إعجاب app.post('/like/:id', express.json(), (req, res) => { const post = posts.find(p => p.id == req.params.id); if (post) { post.likes++; io.emit('like-update', { id: post.id, likes: post.likes }); res.json({ success: true }); } else { res.status(404).json({ error: 'Post not found' }); } }); // WebSocket للشات الفوري (مسنجر + تلغرام) io.on('connection', (socket) => { console.log('مستخدم جديد:', socket.id); socket.on('set-username', (username) => { users[socket.id] = username; io.emit('user-joined', `${username} دخل الدردشة`); // إرسال آخر 50 رسالة للمستخدم الجديد socket.emit('chat-history', chatMessages.slice(-50)); }); socket.on('chat-message', (msg) => { const username = users[socket.id] || 'زائر'; const messageData = { user: username, message: msg, timestamp: new Date().toISOString() }; chatMessages.push(messageData); io.emit('new-chat-message', messageData); }); socket.on('disconnect', () => { const username = users[socket.id]; if (username) { io.emit('user-left', `${username} غادر الدردشة`); delete users[socket.id]; } }); }); server.listen(3000, () => { console.log('🚀 تطبيق السيطرة يعمل على http://localhost:3000'); }); npm install npm start
